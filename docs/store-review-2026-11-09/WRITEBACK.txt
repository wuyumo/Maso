Before copying, from the repo root:

  python3 - << 'PY'
  import hashlib, json, pathlib, shutil, sys
  root = pathlib.Path('.').resolve()
  manifest = json.loads(pathlib.Path('/ABSOLUTE/PATH/masso-review-manifest.json').read_text())
  # Point unpack at the directory where the tarballs were extracted, preserving paths.
  unpack = pathlib.Path('/ABSOLUTE/PATH/unpacked')
  for item in manifest['files']:
      rel = item['path']
      if item.get('packed') is False:
          continue
      src = unpack / rel
      dest = root / rel
      if not src.is_file():
          sys.exit(f'missing in unpack: {rel}')
      new = hashlib.sha256(src.read_bytes()).hexdigest()
      if new != item['newSha256']:
          sys.exit(f'unpack hash mismatch: {rel}')
      if dest.exists():
          live = hashlib.sha256(dest.read_bytes()).hexdigest()
          if item['snapshotSha256'] and live != item['snapshotSha256']:
              sys.exit(f'live file moved, not copying: {rel} live {live}')
          if live == new:
              continue
      elif item['snapshotSha256']:
          sys.exit(f'expected live file missing: {rel}')
      dest.parent.mkdir(parents=True, exist_ok=True)
      shutil.copy2(src, dest)
      print('copied', rel)
  PY
